/**
 * Industry-specific onboarding playbooks for the Kraya account builder.
 *
 * Each industry has 6 step-specific chunks derived from the Phase-2 ICP
 * rule-books (mined from real Kraya client accounts per vertical). The build
 * orchestrator injects only the relevant chunk into each extractor call, and
 * the full set into agent chat context.
 *
 * STYLE NOTE: this file deliberately uses NO em dashes (\u2014) or en dashes (\u2013).
 * Use colons for label separators, periods for sentence joins, and regular
 * hyphens for compound words / time ranges. Reason: this content gets
 * concatenated into every downstream agent's system prompt, and the LLM
 * imitates the punctuation it sees in its system prompt much more strongly
 * than it follows style rules in the user message.
 */

/**
 * Get a specific playbook chunk for a build step.
 */
export function getPlaybookChunk(
  industry: string,
  step: keyof PlaybookChunks
): string | null {
  const playbook = INDUSTRY_PLAYBOOKS[industry as IndustryKey];
  if (!playbook) return null;
  return playbook[step] ?? null;
}

/**
 * Get the full playbook (all chunks concatenated) for injection into agent context.
 */
export function getFullPlaybookForAgents(industry: string): string | null {
  const playbook = INDUSTRY_PLAYBOOKS[industry as IndustryKey];
  if (!playbook) return null;

  return [
    `INDUSTRY: ${industry}`,
    "",
    "--- PIPELINE & STAGES ---",
    playbook.stages,
    "",
    "--- LEAD ATTRIBUTES ---",
    playbook.attributes,
    "",
    "--- QUALIFICATION FLOW ---",
    playbook.orgInfo,
    "",
    "--- SEQUENCES ---",
    playbook.sequences,
    "",
    "--- SMART TRIGGERS ---",
    playbook.rules,
    "",
    "--- FAQ TOPICS ---",
    playbook.faqs,
    "",
    "--- QUICK REPLIES ---",
    playbook.quickReplies,
  ].join("\n");
}

const DETECTION_PROMPT = `Classify this business into exactly one industry key:
- healthcare (clinics, hospitals, dental, dermatology, diagnostics, aesthetic / wellness clinics, mental health)
- realestate (builders, developers, brokers, property agents, plots / commercial leasing)
- fitness (gyms, fitness studios, personal training, yoga / Zumba / dance studios, online transformation coaching)
- travel (travel agencies, tour operators, DMCs, visa / holiday packages, stays / resorts)
- b2b (B2B SERVICES and technology sold to businesses: consulting, IT / software / SaaS, staffing / recruitment, CA / tax / compliance / company-registration, immigration / visa services, other professional services)
- agencies (marketing, creative, digital, branding, content, social-media, PR agencies, and photo / video / event production studios)
- education (coaching centres, test prep, edtech, schools, tutoring, universities, skilling)
- retail (sellers of finished products to CONSUMERS: D2C brands, e-commerce, boutiques, jewellery, apparel, electronics, furniture, food products)
- manufacturing (makers / suppliers of physical goods to BUSINESSES: manufacturers, industrial / building materials, machinery, components, wholesale / bulk B2B product supply)
- other (doesn't clearly fit any above)

Disambiguation:
- retail vs manufacturing: retail sells finished products to end consumers; manufacturing makes or supplies goods B2B (to dealers, businesses, projects), usually with sample / quote / MOQ.
- b2b vs manufacturing: b2b is a SERVICE or software sold to businesses (no physical product); manufacturing sells a physical product B2B.
- b2b vs agencies: agencies are marketing / creative / media services; b2b is every other business-facing service (IT, consulting, compliance, staffing).
- education vs b2b: a study-abroad / immigration consultancy that mainly does visas is b2b; if the core is admissions / coaching / test-prep, choose education.

Reply with ONLY the industry key (one word, lowercase).`;

/**
 * Detect the industry from a client profile text using gpt-4o-mini.
 * Returns one of the SUPPORTED_INDUSTRIES keys, or "other".
 */
export async function detectIndustry(profileText: string): Promise<string> {
  const model = await getAIModel("openai", "gpt-4o-mini", { keyScope: "account-updates" });

  const { text } = await generateText({
    model,
    system: DETECTION_PROMPT,
    prompt: profileText.slice(0, 4000),
    temperature: 0,
    // OpenAI enforces a minimum of 16 here. The classifier only needs to
    // return one word ("fitness", "healthcare", ...) any extra tokens are
    // discarded by the post-processing regex.
    maxOutputTokens: 16,
  });

  const detected = text.trim().toLowerCase().replace(/[^a-z]/g, "");

  if ((SUPPORTED_INDUSTRIES as readonly string[]).includes(detected)) {
    return detected;
  }

  logger.warn("[IndustryPlaybooks] Detection returned unrecognised key, defaulting to 'other'", {
    module: "industry-playbooks",
    raw: text.trim(),
    normalised: detected,
  });

  return "other";
}
